\textbf{Cinemática da partícula} \hfill $\vec r = (P - O)$, parâmetro $u = u(t)$
\[ \vec v = \frac{d\vec r}{du}\,\dot u, \qquad
   \vec a = \frac{d^2\vec r}{du^2}\,\dot u^2 + \frac{d\vec r}{du}\,\ddot u, \qquad
   v = |\vec v|, \qquad \vec v = v\,\vec\tau, \qquad \vec a = \dot v\,\vec\tau + \frac{v^2}{\rho}\,\vec n \]
\[ \vec\tau = \frac{\vec v}{|\vec v|}, \qquad \vec b = \frac{\vec v \wedge \vec a}{|\vec v \wedge \vec a|}, \qquad
   \vec n = \vec b \wedge \vec\tau, \qquad \rho = \frac{|\vec v|^3}{|\vec v \wedge \vec a|}, \qquad
   a_t = \vec a \cdot \vec\tau, \qquad a_n = \vec a \cdot \vec n, \qquad (C_{curv} - P) = \rho\,\vec n \]

\textbf{Teorema de transporte} (derivada em referenciais distintos)
\[ \left.\frac{d\vec u}{dt}\right|_{\mathcal F} = \left.\frac{d\vec u}{dt}\right|_{\mathcal B}
   + \vec\omega_{\mathcal B/\mathcal F} \wedge \vec u \]

\textbf{Cinemática do corpo rígido}
\[ \vec v_P = \vec v_O + \vec\omega \wedge (P - O), \qquad
   \vec a_P = \vec a_O + \dot{\vec\omega} \wedge (P - O) + \vec\omega \wedge [\vec\omega \wedge (P - O)] \]
\[ \text{Propriedade fundamental:} \quad \vec v_A \cdot (B - A) = \vec v_B \cdot (B - A) \]
\[ \text{EHI:} \quad \vec v_{min} = \frac{\vec v_O \cdot \vec\omega}{|\vec\omega|^2}\,\vec\omega, \qquad
   E = O + \frac{\vec\omega \wedge \vec v_O}{|\vec\omega|^2} + \lambda\,\vec\omega \ (\lambda \in \mathbb R), \qquad
   \vec v_P = \vec v_{min} + \vec\omega \wedge (P - E) \]
\[ \text{Movimento plano:} \quad \vec v_{CIR} = \vec 0, \quad \vec v_P = \vec\omega \wedge (P - CIR) \]
\[ \text{Rolamento sem escorregamento:} \quad
   \vec v^{\,(\text{corpo})}_{\text{contato}} = \vec v^{\,(\text{apoio})}_{\text{contato}} \]

\textbf{Composição de movimentos} \hfill (referencial móvel $\mathcal B$, com $\vec\omega_{arr} = \vec\omega_{\mathcal B/\mathcal F}$)
\[ \vec v_P = \vec v_{P,rel} + \vec v_{P,arr}, \qquad \vec v_{P,arr} = \vec v_O + \vec\omega_{arr} \wedge (P - O)
   \quad (O \in \mathcal B) \]
\[ \vec a_P = \vec a_{P,rel} + \vec a_{P,arr} + \vec a_{P,Cor}, \qquad
   \vec a_{P,Cor} = 2\,\vec\omega_{arr} \wedge \vec v_{P,rel} \]
\[ \vec a_{P,arr} = \vec a_O + \dot{\vec\omega}_{arr} \wedge (P - O) + \vec\omega_{arr} \wedge [\vec\omega_{arr} \wedge (P - O)] \]
\[ \vec\omega_{abs} = \vec\omega_{rel} + \vec\omega_{arr}, \qquad
   \dot{\vec\omega}_{abs} = \dot{\vec\omega}_{rel} + \dot{\vec\omega}_{arr} + \vec\omega_{arr} \wedge \vec\omega_{rel} \]

\textbf{Produtos vetoriais da base} \quad
$\vi \wedge \vj = \vk$, \ $\vj \wedge \vk = \vi$, \ $\vk \wedge \vi = \vj$, \
$\vj \wedge \vi = -\vk$, \ $\vk \wedge \vj = -\vi$, \ $\vi \wedge \vk = -\vj$.

\textbf{Valores úteis} \quad
$\sin 30^\circ = \tfrac12$, \ $\cos 30^\circ = \tfrac{\sqrt3}{2}$, \
$\sin 45^\circ = \cos 45^\circ = \tfrac{\sqrt2}{2}$, \ $\sin 60^\circ = \tfrac{\sqrt3}{2}$, \ $\cos 60^\circ = \tfrac12$.
