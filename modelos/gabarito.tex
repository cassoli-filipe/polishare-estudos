% ------------------------------------------------------------------ Questão 1
\questao{Questão 1 (3,0 pontos)}
\begin{itens}
  \item $\vec v = 2\,\vi + 2t\,\vj + t^2\,\vk$ e $\vec a = 2\,\vj + 2t\,\vk$. Em $t = 1$~s:
    \quad $\res{\vec v = 2\,\vi + 2\,\vj + \vk \ \text{(m/s)}}$ \quad $\res{\vec a = 2\,\vj + 2\,\vk \ \text{(m/s}^2\text{)}}$
  \item $|\vec v|^2 = 4 + 4t^2 + t^4 = (t^2 + 2)^2$, então $\res{|\vec v| = t^2 + 2 \ \text{(m/s)}}$ e
    $a_t = \dfrac{d|\vec v|}{dt} = 2t$, ou seja, $\res{a_t = 2 \ \text{m/s}^2}$ em $t = 1$~s.
    Pelo outro caminho, $a_t = \vec a \cdot \vec\tau = (0 + 4 + 2)/3 = 2$. Em $t = 1$~s, $|\vec v| = 3$~m/s.
  \item $\res{\vec\tau = \dfrac{\vec v}{|\vec v|} = \dfrac{2\,\vi + 2\,\vj + \vk}{3}}$. A parcela normal é
    $a_n\,\vec n = \vec a - a_t\,\vec\tau = -\tfrac43\,\vi + \tfrac23\,\vj + \tfrac43\,\vk$, de módulo
    $a_n = \sqrt{16 + 4 + 16}/3 = 2$, então
    $\res{\vec n = \dfrac{-2\,\vi + \vj + 2\,\vk}{3}}$ e $\res{\vec b = \vec\tau \wedge \vec n = \dfrac{\vi - 2\,\vj + 2\,\vk}{3}}$.
    Conferência: $\vec v \wedge \vec a = 2\,\vi - 4\,\vj + 4\,\vk$, na direção de $\vec b$\ok.
  \item $\res{\vec v = 3\,\vec\tau \ \text{(m/s)}}$ e $\res{\vec a = 2\,\vec\tau + 2\,\vec n \ \text{(m/s}^2\text{)}}$.
    A componente binormal é nula.
  \item $\res{\rho = \dfrac{|\vec v|^2}{a_n} = \dfrac92 = 4{,}5 \ \text{m}}$. Pela fórmula,
    $\rho = |\vec v|^3/|\vec v \wedge \vec a| = 27/6 = 4{,}5$~m\ok.
\end{itens}
\criterios{
  a) & $\vec v$ 0,25; $\vec a$ 0,25 \\
  b) & $|\vec v(t)|$ 0,25; $a_t$ 0,25. Aceite $a_t = \vec a \cdot \vec\tau$ sem $|\vec v(t)|$ só se $|\vec v| = 3$ estiver certo. \\
  c) & $\vec\tau$ 0,25; $\vec n$ 0,5; $\vec b$ 0,25 \\
  d) & $\vec v$ 0,25; $\vec a$ 0,25 \\
  e) & $\rho$ 0,5 \\
}
\inspirada{2024 Q1 (curva 3D, Frenet e $\rho = |\vec v|^3/|\vec v \wedge \vec a|$), 2023 Q1 ($a_t \neq 0$,
$\vec a = a_t\vec\tau + a_n\vec n$) e 2022R Q1 ($\vec b = \vec\tau \wedge \vec n$). Diferente de 2024 e
2022R, aqui $a_t \neq 0$: $\vec n$ não sai direto de $\vec a/|\vec a|$.}

% ------------------------------------------------------------------ Questão 2
\questao{Questão 2 (3,5 pontos)}
Origem no ponto de contato do disco com o piso: $C = (0, R)$, $A = (0, 2R)$, $B = (2R, 4R)$.
\begin{itens}
  \item O CIR do disco é o ponto de contato $I_d$ (rolamento sem escorregamento):
    $\vec v_A = -\omega\,\vk \wedge 2R\,\vj$, ou seja, $\res{\vec v_A = 2\omega R\,\vi}$. Como $\omega$ é
    constante, $\vec\alpha_{\text{disco}} = \vec 0$ e $\vec a_C = \vec 0$, logo $\vec a_A = -\omega^2 (A - C)$:
    $\res{\vec a_A = -\omega^2 R\,\vj}$ (aponta para o centro).
  \item $\vec v_A$ é horizontal, então o CIR está na \textbf{vertical por $A$}; $\vec v_B$ é vertical (guia),
    então o CIR está na \textbf{horizontal por $B$}. O cruzamento é $\text{CIR}_{AB} = (0, 4R)$:
    $\res{(\text{CIR} - A) = 2R\,\vj}$. No esboço, as perpendiculares às velocidades de $A$ e de $B$ se
    cruzam acima de $A$, na altura de $B$.
  \item $\vec v_A = \vec\omega_{AB} \wedge (A - \text{CIR}) = \omega_{AB}\,\vk \wedge (-2R\,\vj) = 2R\,\omega_{AB}\,\vi$.
    Igualando a $2\omega R\,\vi$: $\res{\vec\omega_{AB} = \omega\,\vk}$ (anti-horário). Então
    $\vec v_B = \omega\,\vk \wedge 2R\,\vi$, ou seja, $\res{\vec v_B = 2\omega R\,\vj}$ (o cursor sobe).
    Conferência pelas projeções: $\vec v_A \cdot (B - A) = 4\omega R^2 = \vec v_B \cdot (B - A)$\ok.
  \item Com $\vec a_B = a_B\,\vj$ (guia vertical):
    \[ \vec a_B = \vec a_A + \alpha_{AB}\,\vk \wedge (2R\,\vi + 2R\,\vj) - \omega^2 (2R\,\vi + 2R\,\vj)
       = (-2R\alpha_{AB} - 2\omega^2 R)\,\vi + (-\omega^2 R + 2R\alpha_{AB} - 2\omega^2 R)\,\vj \]
    A componente em $\vi$ tem que ser nula: $\res{\vec\alpha_{AB} = -\omega^2\,\vk}$. Na componente em $\vj$:
    $\res{\vec a_B = -5\omega^2 R\,\vj}$.
\end{itens}
\criterios{
  a) & $\vec v_A$ 0,25; $\vec a_A$ 0,25. Zero em $\vec a_A$ se usar $\vec a_A = \vec 0$ ``porque $\omega$ é constante''. \\
  b) & desenho com as duas perpendiculares 0,5; $(\text{CIR} - A)$ 0,5 \\
  c) & $\vec\omega_{AB}$ 0,5 (sinal incluso); $\vec v_B$ 0,5 \\
  d) & equação vetorial montada com o vínculo da guia 0,3; $\vec\alpha_{AB}$ 0,35; $\vec a_B$ 0,35 \\
}
\inspirada{2024R Q3 (disco com $\omega$ constante rolando, ponto mais alto, barra, CIR por esboço e $\alpha$
pela barra), 2023 Q2 (rolamento + guia; CIR no cruzamento da vertical com a horizontal) e 2025 Q2d
(aceleração de ponto de barra).}

% ------------------------------------------------------------------ Questão 3
\questao{Questão 3 (3,5 pontos)}
Referencial móvel: o braço ($Oxyz$). Então $\vec\omega_{\text{arr}} = \Omega\,\vk$,
$\vec\alpha_{\text{arr}} = \vec 0$, $\vec v_O = V\,\vk$ e $\vec a_O = \vec 0$, com $(P - O) = L\,\vi + R\,\vk$ e
$(P - A) = R\,\vk$.
\begin{itens}
  \item $\res{\vec v_{\text{rel}} = \omega\,\vj \wedge R\,\vk = \omega R\,\vi}$ \quad
    $\res{\vec v_{\text{arr}} = V\,\vk + \Omega\,\vk \wedge (L\,\vi + R\,\vk) = \Omega L\,\vj + V\,\vk}$ \\[3pt]
    $\res{\vec v_P = \omega R\,\vi + \Omega L\,\vj + V\,\vk}$
  \item $\res{\vec a_{\text{rel}} = \dot\omega\,\vj \wedge R\,\vk + \omega\,\vj \wedge (\omega\,\vj \wedge R\,\vk) = \dot\omega R\,\vi - \omega^2 R\,\vk}$ \\[3pt]
    $\res{\vec a_{\text{arr}} = \Omega\,\vk \wedge [\Omega\,\vk \wedge (L\,\vi + R\,\vk)] = -\Omega^2 L\,\vi}$
    (o $V$ constante não contribui) \\[3pt]
    $\res{\vec a_{\text{Cor}} = 2\,\Omega\,\vk \wedge \omega R\,\vi = 2\,\Omega\omega R\,\vj}$ \\[3pt]
    $\res{\vec a_P = (\dot\omega R - \Omega^2 L)\,\vi + 2\,\Omega\omega R\,\vj - \omega^2 R\,\vk}$
  \item $\res{\vec\omega = \vec\omega_{\text{rel}} + \vec\omega_{\text{arr}} = \omega\,\vj + \Omega\,\vk}$
  \item $\res{\vec\alpha = \dot\omega\,\vj + \vec 0 + \Omega\,\vk \wedge \omega\,\vj = -\Omega\omega\,\vi + \dot\omega\,\vj}$
\end{itens}
Conferência pelo campo de acelerações do disco: com $\vec v_A = V\,\vk + \Omega L\,\vj$ e
$\vec a_A = -\Omega^2 L\,\vi$, a expressão
$\vec a_P = \vec a_A + \vec\alpha \wedge R\,\vk + \vec\omega \wedge (\vec\omega \wedge R\,\vk)$ dá a mesma
resposta do item b\ok.
\criterios{
  a) & $\vec v_{\text{rel}}$ 0,4; $\vec v_{\text{arr}}$ 0,4 (vale 0,2 se esquecer $V\,\vk$); $\vec v_{\text{abs}}$ 0,2 \\
  b) & $\vec a_{\text{rel}}$ 0,4 (0,2 sem o termo $\dot\omega R\,\vi$); $\vec a_{\text{arr}}$ 0,4; $\vec a_{\text{Cor}}$ 0,4 (inclui o fator 2); $\vec a_{\text{abs}}$ 0,3 \\
  c) & $\vec\omega$ 0,5 \\
  d) & $\vec\alpha$ 0,5 (vale 0,25 sem o termo $\vec\omega_{\text{arr}} \wedge \vec\omega_{\text{rel}}$) \\
}
\inspirada{2025 Q3 (garfo que gira e translada; o arrastamento inclui a translação da origem), 2022 Q1
(cilindro que gira e sobe, disco num braço), 2022R Q3 (plataforma + braço + disco com $P$ no ponto mais
alto) e 2024 Q3 ($\alpha$ absoluto com $\vec\omega_{\text{arr}} \wedge \vec\omega_{\text{rel}}$).}
